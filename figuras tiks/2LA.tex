\usepackage{tikz}

\begin{document}

\begin{tikzpicture}[scale=1.2, >=Stealth]
    % Líneas horizontales de niveles de energía
    \draw[thick] (-4,0) -- (1,0) node[right] {};
    \draw[thick] (-4,6) -- (1,6) node[right] {};

    % Línea punteada de desintonía (paralela al estado excitado)
    \draw[dashed, thick] (-4,7) -- (1,7);

    % Flecha de medida de la desintonía Δ
    \draw[<->, thick] (1.3,6) -- (1.3,7)
        node[midway, right, font=\Large\bfseries] {$\Delta$};

    % Flecha Ω (acoplamiento)
    \draw[cyan, ultra thick, <->] (-1,0.2) -- (-1,5.8) 
        node[midway, right, green!70!black, font=\Large\bfseries] {$\Omega$};

    % Decaimiento γ (de |e> a |g>)
    \draw[red, dashed, thick, ->, decorate, decoration={snake, amplitude=0.1cm, segment length=1cm}] 
        (-3,6) -- (-3,0.2)
        node[midway, left, red,font=\Large\bfseries] {$\gamma$};

    % Etiquetas de los estados
    \node[left,font=\large\bfseries] at (-4,6) {$|e\rangle$};
    \node[left,font=\large\bfseries] at (-4,0) {$|g\rangle$};
